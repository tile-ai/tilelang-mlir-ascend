# Ascend 910 (A3) backend: independent of the Ascend 950 sources.

file(GLOB TILE_LANG_ASCEND910_ALWAYS_SRCS
  ${CMAKE_CURRENT_LIST_DIR}/target.cc
)
list(APPEND TILE_LANG_SRCS ${TILE_LANG_ASCEND910_ALWAYS_SRCS})

option(USE_ASCEND910 "Build the Ascend 910 (A3) MLIR code generator" OFF)
if(NOT USE_ASCEND910)
  return()
endif()

if(NOT BISHENGIR_ROOT_PATH)
  message(FATAL_ERROR "USE_ASCEND910=ON requires -DBISHENGIR_ROOT_PATH=<AscendNPU-IR install>")
endif()

message(STATUS "Ascend 910 (A3) backend is enabled: ${BISHENGIR_ROOT_PATH}")
include_directories(${BISHENGIR_ROOT_PATH}/include)

file(GLOB TILE_LANG_ASCEND910_ACTIVE_SRCS
  ${CMAKE_CURRENT_LIST_DIR}/op/*.cc
  ${CMAKE_CURRENT_LIST_DIR}/op/*.cpp
  ${CMAKE_CURRENT_LIST_DIR}/target/*.cc
  ${CMAKE_CURRENT_LIST_DIR}/target/*.cpp
  ${CMAKE_CURRENT_LIST_DIR}/transform/*.cc
)
list(APPEND TILE_LANG_SRCS ${TILE_LANG_ASCEND910_ACTIVE_SRCS})
list(REMOVE_DUPLICATES TILE_LANG_SRCS)

file(GLOB TILELANG_ASCEND910_NPUIR_LIBS "${BISHENGIR_ROOT_PATH}/lib/*.a")
set(TILELANG_ASCEND910_NPUIR_LIBS ${TILELANG_ASCEND910_NPUIR_LIBS} PARENT_SCOPE)
set(TILELANG_ASCEND910_ENABLED ON)
set(TILELANG_ASCEND910_ENABLED ON PARENT_SCOPE)
link_directories(${BISHENGIR_ROOT_PATH}/lib)
